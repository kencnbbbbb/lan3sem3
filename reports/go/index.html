const (
        ArrayMinCap = 16
        ArrayMaxCap = 256 * 256
)

type Array struct {
        data []string
        size int
}

func NewArray(capacity int) (*Array, error) {
        if capacity > ArrayMaxCap {
                return nil, fmt.Errorf("%w: %d", ErrCapacity, capacity)
        }
        if capacity < ArrayMinCap {
                capacity = ArrayMinCap
        }
        return &Array{data: make([]string, capacity)}, nil
}

func (a *Array) grow() error {
        newCap := len(a.data) * 2
        if newCap == 0 {
                newCap = ArrayMinCap
        }
        if newCap > ArrayMaxCap {
                return ErrCapacity
        }
        fresh := make([]string, newCap)
        copy(fresh, a.data[:a.size])
        a.data = fresh
        return nil
}

func (a *Array) ensureSpace() error {
        if a.size >= len(a.data) {
                return a.grow()
        }
        return nil
}

func (a *Array) checkIndex(idx int) error {
        if idx < 0 || idx >= a.size {
                return fmt.Errorf("%w: %d (размер %d)", ErrOutOfRange, idx, a.size)
        }
        return nil
}

func (a *Array) PushBack(v string) error {
        if err := a.ensureSpace(); err != nil {
                return err
        }
        a.data[a.size] = v
        a.size++
        return nil
}

func (a *Array) Insert(idx int, v string) error {
        if idx < 0 || idx > a.size {
                return fmt.Errorf("%w: %d (размер %d)", ErrOutOfRange, idx, a.size)
        }
        if err := a.ensureSpace(); err != nil {
                return err
        }
        copy(a.data[idx+1:a.size+1], a.data[idx:a.size])
        a.data[idx] = v
        a.size++
        return nil
}

func (a *Array) Get(idx int) (string, error) {
        if err := a.checkIndex(idx); err != nil {
                return "", err
        }
        return a.data[idx], nil
}

        }

        return nil

func HashDJB2(key string) uint64 {
        var h uint64 = 5381
        for i := 0; i < len(key); i++ {
                h = (h << 5) + h + uint64(key[i])
        }
        return h
}

func HashFNV1a(key string) uint64 {
        var h uint64 = 14695981039346656037
        for i := 0; i < len(key); i++ {
                h ^= uint64(key[i])
                h *= 1099511628211
        }
        return h
}

const (
        OpenHashMinCap = 16
        OpenHashMaxCap = 256 * 256
        openMaxLoad    = 0.7
)

type slotStatus uint8

const (
        slotEmpty slotStatus = iota
        slotUsed
        slotDeleted
)

type slot struct {
        status slotStatus
        val    string
}

type OpenHash struct {
        linear  bool
        slots   []slot
        size    int
        deleted int
}

func NewOpenHash(capacity int) (*OpenHash, error) {
        if capacity > OpenHashMaxCap {
                return nil, ErrCapacity
        }
        c := OpenHashMinCap
        for c < capacity {
                c <<= 1
        }
        return &OpenHash{slots: make([]slot, c)}, nil
}

func (h *OpenHash) probe(key string, i uint64) int {
        mask := uint64(len(h.slots) - 1)
        step := HashFNV1a(key) | 1
        if h.linear {
                step = 1
        }
        return int((HashDJB2(key) + i*step) & mask)
}

func (h *OpenHash) Find(value string) int {
        for i := range uint64(len(h.slots)) {
                idx := h.probe(value, i)
                switch {
                case h.slots[idx].status == slotEmpty:
                        return -1
                case h.slots[idx].status == slotUsed && h.slots[idx].val == value:
                        return idx
                }
        }
        return -1
}

func (h *OpenHash) Contains(value string) bool { return h.Find(value) != -1 }

func (h *OpenHash) place(value string) {
        for i := uint64(0); ; i++ {
                s := &h.slots[h.probe(value, i)]
                if s.status != slotUsed {
                        if s.status == slotDeleted {
                                h.deleted--
                        }
                        *s = slot{status: slotUsed, val: value}
                        h.size++
                        return
                }
        }
}

func (h *OpenHash) rehash(newCap int) error {
        if newCap > OpenHashMaxCap {
                return ErrCapacity
        }
        old := h.slots
        h.slots = make([]slot, newCap)
        h.size, h.deleted = 0, 0
        for _, s := range old {
                if s.status == slotUsed {
                        h.place(s.val)
                }
        }
        return nil
}

func (h *OpenHash) Insert(value string) (bool, error) {
        if h.Contains(value) {
                return false, nil
        }
        if len(h.slots) == 0 {
                h.slots = make([]slot, OpenHashMinCap)
        }
        capacity := float64(len(h.slots))
        if float64(h.size+h.deleted+1) > capacity*openMaxLoad {
                newCap := len(h.slots)
                if float64(h.size+1) > capacity*openMaxLoad/2 {
                        newCap *= 2
                }
                if err := h.rehash(newCap); err != nil {
                        return false, err
                }
        }
        h.place(value)
        return true, nil
}

func (h *OpenHash) Erase(value string) bool {
        idx := h.Find(value)
        if idx == -1 {
                return false
        }
        h.slots[idx] = slot{status: slotDeleted}
        h.size--
        h.deleted++
        return true
}

func (h *OpenHash) Clear() {
        for i := range h.slots {
                h.slots[i] = slot{}
        }
        h.size, h.deleted = 0, 0
}

func (h *OpenHash) Len() int { return h.size }

func (h *OpenHash) Cap() int { return len(h.slots) }

func (h *OpenHash) Empty() bool { return h.size == 0 }

func (h *OpenHash) Items() []string {
        out := make([]string, 0, h.size)
        for _, s := range h.slots {
                if s.status == slotUsed {
                        out = append(out, s.val)
                }
        }
        return out
}

func (h *OpenHash) String() string { return joinWithSpaces(h.Items()) }

func (h *OpenHash) Clone() *OpenHash {
        c := &OpenHash{slots: make([]slot, len(h.slots)), size: h.size, deleted: h.deleted, linear: h.linear}
        copy(c.slots, h.slots)
        return c
}

func (h *OpenHash) replace(items []string) error {
        fresh := &OpenHash{linear: h.linear}
        for _, v := range items {
                if _, err := fresh.Insert(v); err != nil {
                        return err
                }
        }
        *h = *fresh
        return nil
}

}

func (h *OpenHash) DeserializeBinary(r io.Reader) error {
        items, err := deserializeStringsBinary(r, h.binaryTag())

        }

        var prev *oneNode

                }

        del := l.head
        if prev != nil {
                del = prev.next
                prev.next = del.next
        } else {
                l.head = del.next
        }
        if del == l.tail {
                l.tail = prev
        }
        l.size--
        return del.data
}

const MaxItems = 1 << 24

type Serializer interface {
        SerializeBinary(w io.Writer) error
        DeserializeBinary(r io.Reader) error
        SerializeText(w io.Writer) error
        DeserializeText(r *bufio.Reader) error
}

func writeTag(w io.Writer, tag string) error {
        n, err := io.WriteString(w, tag)
        if err == nil && n != len(tag) {
                return io.ErrShortWrite
        }
        return err
}

func readTag(r io.Reader, tag string) error {
        buf := make([]byte, 4)
        if _, err := io.ReadFull(r, buf); err != nil {
                return fmt.Errorf("%w: тег: %w", ErrFormat, err)
        }
        if string(buf) != tag {
                return fmt.Errorf("%w: ожидался тег %s, получен %q", ErrFormat, tag, buf)
        }
        return nil
}

func writeSize(w io.Writer, n int) error {
        return binary.Write(w, binary.LittleEndian, uint64(n))
}

func readSize(r io.Reader) (int, error) {
        var n uint64
        if err := binary.Read(r, binary.LittleEndian, &n); err != nil {
                return 0, fmt.Errorf("%w: размер: %w", ErrFormat, err)
        }
        if n > MaxItems {
                return 0, fmt.Errorf("%w: слишком большой размер %d", ErrFormat, n)
        }
        return int(n), nil
}

func writeString(w io.Writer, s string) error {
        if err := writeSize(w, len(s)); err != nil {
                return err
        }
        written, err := io.WriteString(w, s)
        if err == nil && written != len(s) {
                return io.ErrShortWrite
        }
        return err
}

func readString(r io.Reader) (string, error) {
        n, err := readSize(r)
        if err != nil {
                return "", err
        }
        buf := make([]byte, n)
        if _, err := io.ReadFull(r, buf); err != nil {
                return "", fmt.Errorf("%w: строка: %w", ErrFormat, err)
        }
        return string(buf), nil
}

func writeInt(w io.Writer, v int) error {
        if v < math.MinInt32 || v > math.MaxInt32 {
                return fmt.Errorf("%w: число %d не помещается в int32", ErrFormat, v)
        }
        return binary.Write(w, binary.LittleEndian, int32(v))
}

func readInt(r io.Reader) (int, error) {
        var v int32
        if err := binary.Read(r, binary.LittleEndian, &v); err != nil {
                return 0, fmt.Errorf("%w: число: %w", ErrFormat, err)
        }
        return int(v), nil
}

func readToken(r *bufio.Reader) (string, error) {
        var sb strings.Builder
        for {
                c, _, err := r.ReadRune()
                if err != nil {
                        if sb.Len() > 0 && errors.Is(err, io.EOF) {
                                return sb.String(), nil
                        }
                        return "", fmt.Errorf("%w: ожидалось слово: %w", ErrFormat, err)
                }
                if unicode.IsSpace(c) {
                        if sb.Len() > 0 {
                                return sb.String(), nil
                        }
                        continue
                }
                sb.WriteRune(c)
        }
}

func writeTextHeader(w io.Writer, tag string, n int) error {
        _, err := fmt.Fprintf(w, "%s %d\n", tag, n)
        return err
}

func readTextHeader(r *bufio.Reader, tag string) (int, error) {
        got, err := readToken(r)
        if err != nil {
                return 0, err
        }
        if got != tag {
                return 0, fmt.Errorf("%w: ожидался заголовок %s, получен %q", ErrFormat, tag, got)
        }
        n, err := readTextInt(r)
        if err != nil {
                return 0, err
        }
        if n < 0 || n > MaxItems {
                return 0, fmt.Errorf("%w: недопустимый размер %d", ErrFormat, n)
        }
        return n, nil
}

func readTextInt(r *bufio.Reader) (int, error) {
        tok, err := readToken(r)
        if err != nil {
                return 0, err
        }
        v, err := strconv.ParseInt(tok, 10, 32)
        if err != nil {
                return 0, fmt.Errorf("%w: не число %q", ErrFormat, tok)
        }
        return int(v), nil
}

func PeekTextTag(r *bufio.Reader) (string, error) {
        buf, _ := r.Peek(64)
        i := 0
        for i < len(buf) && unicode.IsSpace(rune(buf[i])) {
                i++
        }
        j := i
        for j < len(buf) && !unicode.IsSpace(rune(buf[j])) {
                j++
        }
        if i == j {
                return "", fmt.Errorf("%w: ожидался тег структуры", ErrFormat)
        }
        return string(buf[i:j]), nil
}

func PeekBinaryTag(r *bufio.Reader) (string, error) {
        buf, err := r.Peek(4)
        if err != nil {
                return "", fmt.Errorf("%w: тег структуры: %w", ErrFormat, err)
        }
        return string(buf), nil
}

func WriteTag(w io.Writer, tag string) error { return writeTag(w, tag) }

func ReadTag(r io.Reader, tag string) error { return readTag(r, tag) }

func WriteSize(w io.Writer, n int) error { return writeSize(w, n) }

func ReadSize(r io.Reader) (int, error) { return readSize(r) }

func WriteString(w io.Writer, s string) error { return writeString(w, s) }

func ReadString(r io.Reader) (string, error) { return readString(r) }

func WriteTextHeader(w io.Writer, tag string, n int) error { return writeTextHeader(w, tag, n) }

func ReadTextHeader(r *bufio.Reader, tag string) (int, error) { return readTextHeader(r, tag) }

func WriteTextString(w io.Writer, s string) error { return writeTextString(w, s) }

func ReadTextString(r *bufio.Reader) (string, error) { return readTextString(r) }

func writeTextString(w io.Writer, s string) error {
        var sb strings.Builder
        sb.WriteByte('"')
        for i := 0; i < len(s); i++ {
                if s[i] == '"' || s[i] == '\\' {
                        sb.WriteByte('\\')
                }
                sb.WriteByte(s[i])
        }
        sb.WriteString("\"\n")
        _, err := io.WriteString(w, sb.String())
        return err
}

func readTextString(r *bufio.Reader) (string, error) {
        for {
                b, err := r.ReadByte()
                if err != nil {
                        return "", fmt.Errorf("%w: ожидалась строка: %w", ErrFormat, err)
                }
                if b == '"' {
                        break
                }
                if !unicode.IsSpace(rune(b)) {
                        return "", fmt.Errorf("%w: строка должна начинаться с кавычки", ErrFormat)
                }
        }
        var sb strings.Builder
        for {
                b, err := r.ReadByte()
                if err != nil {
                        return "", fmt.Errorf("%w: незакрытая строка: %w", ErrFormat, err)
                }
                switch b {
                case '"':
                        return sb.String(), nil
                case '\\':
                        next, err := r.ReadByte()
                        if err != nil {
                                return "", fmt.Errorf("%w: обрыв после \\: %w", ErrFormat, err)
                        }
                        sb.WriteByte(next)
                default:
                        sb.WriteByte(b)
                }
        }
}

func writeTextInt(w io.Writer, v int) error {
        _, err := fmt.Fprintf(w, "%d\n", v)
        return err
}

func serializeStringsBinary(w io.Writer, tag string, items []string) error {
        if err := writeTag(w, tag); err != nil {
                return err
        }
        if err := writeSize(w, len(items)); err != nil {
                return err
        }
        for _, s := range items {
                if err := writeString(w, s); err != nil {
                        return err
                }
        }
        return nil
}

func deserializeStringsBinary(r io.Reader, tag string) ([]string, error) {
        if err := readTag(r, tag); err != nil {
                return nil, err
        }
        n, err := readSize(r)
        if err != nil {
                return nil, err
        }
        items := make([]string, 0, n)
        for i := 0; i < n; i++ {
                s, err := readString(r)
                if err != nil {
                        return nil, err
                }
                items = append(items, s)
        }
        return items, nil
}

func serializeStringsText(w io.Writer, tag string, items []string) error {
        if err := writeTextHeader(w, tag, len(items)); err != nil {
                return err
        }
        for _, s := range items {
                if err := writeTextString(w, s); err != nil {
                        return err
                }
        }
        return nil
}

func deserializeStringsText(r *bufio.Reader, tag string) ([]string, error) {
        n, err := readTextHeader(r, tag)
        if err != nil {
                return nil, err
        }
        items := make([]string, 0, n)
        for i := 0; i < n; i++ {
                s, err := readTextString(r)
                if err != nil {
                        return nil, err
                }
                items = append(items, s)
        }
        return items, nil
}

type Stack struct {
        head *oneNode
        size int
}

func (s *Stack) Push(v string) {
        s.head = &oneNode{data: v, next: s.head}
        s.size++
}

func (s *Stack) Pop() (string, error) {
        if s.head == nil {
                return "", ErrEmpty
        }
        v := s.head.data
        s.head = s.head.next
        s.size--
        return v, nil
}

func (s *Stack) Top() (string, error) {
        if s.head == nil {
                return "", ErrEmpty
        }
        return s.head.data, nil
}

func (s *Stack) Clear() { *s = Stack{} }

func (s *Stack) Len() int { return s.size }

func (s *Stack) Empty() bool { return s.head == nil }

func (s *Stack) Items() []string {
        out := make([]string, 0, s.size)
        for node := s.head; node != nil; node = node.next {
                out = append(out, node.data)
        }
        return out
}

func (s *Stack) String() string { return joinWithSpaces(s.Items()) }

func (s *Stack) Clone() *Stack {
        c := &Stack{}
        c.fromBottom(reversed(s.Items()))
        return c
}

func (s *Stack) fromBottom(items []string) {
        s.Clear()
        for _, v := range items {
                s.Push(v)
        }
}

func (s *Stack) SerializeBinary(w io.Writer) error {
        return serializeStringsBinary(w, "STAK", reversed(s.Items()))
}

func (s *Stack) DeserializeBinary(r io.Reader) error {
        items, err := deserializeStringsBinary(r, "STAK")
        if err != nil {
                return err
        }
        s.fromBottom(items)
        return nil
}

func (s *Stack) SerializeText(w io.Writer) error {
        return serializeStringsText(w, "Stack", reversed(s.Items()))
}

func (s *Stack) DeserializeText(r *bufio.Reader) error {
        items, err := deserializeStringsText(r, "Stack")
        if err != nil {
                return err
        }
        s.fromBottom(items)
        return nil
}

type Queue struct {
        head *oneNode
        tail *oneNode
        size int
}

func (q *Queue) Push(v string) {
        node := &oneNode{data: v}
        if q.tail != nil {
                q.tail.next = node
        } else {
                q.head = node
        }
        q.tail = node
        q.size++
}

func (q *Queue) Pop() (string, error) {
        if q.head == nil {
                return "", ErrEmpty
        }
        v := q.head.data
        q.head = q.head.next
        if q.head == nil {
                q.tail = nil
        }
        q.size--
        return v, nil
}

func (q *Queue) Front() (string, error) {
        if q.head == nil {
                return "", ErrEmpty
        }
        return q.head.data, nil
}

func (q *Queue) Clear() { *q = Queue{} }

func (q *Queue) Len() int { return q.size }

func (q *Queue) Empty() bool { return q.head == nil }

func (q *Queue) Items() []string {
        out := make([]string, 0, q.size)
        for node := q.head; node != nil; node = node.next {
                out = append(out, node.data)
        }
        return out
}

func (q *Queue) String() string { return joinWithSpaces(q.Items()) }

func (q *Queue) Clone() *Queue {
        c := &Queue{}
        c.replace(q.Items())
        return c
}

func (q *Queue) replace(items []string) {
        q.Clear()
        for _, v := range items {
                q.Push(v)
        }
}

func (q *Queue) SerializeBinary(w io.Writer) error {
        return serializeStringsBinary(w, "QUEU", q.Items())
}

func (q *Queue) DeserializeBinary(r io.Reader) error {
        items, err := deserializeStringsBinary(r, "QUEU")
        if err != nil {
                return err
        }
        q.replace(items)
        return nil
}

func (q *Queue) SerializeText(w io.Writer) error {
        return serializeStringsText(w, "Queue", q.Items())
}

func (q *Queue) DeserializeText(r *bufio.Reader) error {
        items, err := deserializeStringsText(r, "Queue")
        if err != nil {
                return err
        }
        q.replace(items)
        return nil
}

type treeNode struct {
        data        int
        left, right *treeNode
}
type Tree struct {
        root *treeNode
        size int
}

func (t *Tree) Insert(v int) bool {
        if v < math.MinInt32 || v > math.MaxInt32 {
                return false
        }
        n := &treeNode{data: v}
        if t.root == nil {
                t.root = n
                t.size++
                return true
        }
        queue := []*treeNode{t.root}
        for i := 0; i < len(queue); i++ {
                cur := queue[i]
                if cur.left == nil {
                        cur.left = n
                        t.size++
                        return true
                }
                if cur.right == nil {
                        cur.right = n
                        t.size++
                        return true
                }
                queue = append(queue, cur.left, cur.right)
        }
        return false
}
func (t *Tree) IsMember(v int) bool {
        queue := []*treeNode{}
        if t.root != nil {
                queue = append(queue, t.root)
        }
        for i := 0; i < len(queue); i++ {
                n := queue[i]
                if n.data == v {
                        return true
                }
                if n.left != nil {
                        queue = append(queue, n.left)
                }
                if n.right != nil {
                        queue = append(queue, n.right)
                }
        }
        return false
}
func (t *Tree) IsComplete() bool {
        queue := []*treeNode{t.root}
        gap := false
        for i := 0; i < len(queue); i++ {
                n := queue[i]
                if n == nil {
                        gap = true
                        continue
                }
                if gap {
                        return false
                }
                queue = append(queue, n.left, n.right)
        }
        return true
}
func (t *Tree) Clear()      { *t = Tree{} }
func (t *Tree) Len() int    { return t.size }
func (t *Tree) Empty() bool { return t.root == nil }
func (t *Tree) Height() int {
        n := t.size
        h := 0
        for n > 0 {
                h++
                n >>= 1
        }
        return h
}
func (t *Tree) Items() []int {
        out := make([]int, 0, t.size)
        queue := []*treeNode{}
        if t.root != nil {
                queue = append(queue, t.root)
        }
        for i := 0; i < len(queue); i++ {
                n := queue[i]
                out = append(out, n.data)
                if n.left != nil {
                        queue = append(queue, n.left)
                }
                if n.right != nil {
                        queue = append(queue, n.right)
                }
        }
        return out
}
func (t *Tree) String() string {
        var sb strings.Builder
        for _, v := range t.Items() {
                sb.WriteString(" ")
                sb.WriteString(strconv.Itoa(v))
        }
        return sb.String()
}
func (t *Tree) Structure() string {
        var sb strings.Builder
        type item struct {
                node *treeNode
                path string
        }
        queue := []item{}
        if t.root != nil {
                queue = append(queue, item{t.root, "root"})
        }
        for i := 0; i < len(queue); i++ {
                it := queue[i]
                fmt.Fprintf(&sb, "%s = %d\n", it.path, it.node.data)
                if it.node.left != nil {
                        queue = append(queue, item{it.node.left, it.path + "->left"})
                }
                if it.node.right != nil {
                        queue = append(queue, item{it.node.right, it.path + "->right"})
                }
        }
        return sb.String()
}
func (t *Tree) levelOrder() []int { return t.Items() }
func (t *Tree) Clone() *Tree {
        c := &Tree{}
        for _, v := range t.Items() {
                c.Insert(v)
        }
        return c
}
func (t *Tree) replace(values []int) {
        fresh := Tree{}
        for _, v := range values {
                fresh.Insert(v)
        }
        *t = fresh
}
func (t *Tree) SerializeBinary(w io.Writer) error {
        if err := writeTag(w, "CTRE"); err != nil {
                return err
        }
        if err := writeSize(w, t.size); err != nil {
                return err
        }
        for _, v := range t.levelOrder() {
                if err := writeInt(w, v); err != nil {
                        return err
                }
        }
        return nil
}

func (t *Tree) DeserializeBinary(r io.Reader) error {
        if err := readTag(r, "CTRE"); err != nil {
                return err
        }
        n, err := readSize(r)
        if err != nil {
                return err
        }
        values := make([]int, 0, n)
        for i := 0; i < n; i++ {
                v, err := readInt(r)
                if err != nil {
                        return err
                }
                values = append(values, v)
        }
        t.replace(values)
        return nil
}

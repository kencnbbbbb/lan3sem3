cmake_minimum_required(VERSION 3.20)
project(laba3_variant5 LANGUAGES CXX)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(LAB3_COVERAGE "Clang source coverage" OFF)
option(LAB3_SANITIZERS "Address and undefined behavior sanitizers" OFF)
set(LAB3_SANITIZER_SET "address,undefined" CACHE STRING "Clang sanitizer list")
set(LAB3_BOOST_ROOT "" CACHE PATH "Optional Boost source tree")
include(FetchContent)
set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
set(BENCHMARK_ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(BENCHMARK_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
set(CATCH_INSTALL_DOCS OFF CACHE BOOL "" FORCE)
set(CATCH_INSTALL_EXTRAS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(googletest GIT_REPOSITORY https://github.com/google/googletest.git GIT_TAG 52eb8108c5bdec04579160ae17225d66034bd723)
FetchContent_Declare(catch2 GIT_REPOSITORY https://github.com/catchorg/Catch2.git GIT_TAG 2b60af89e23d28eefc081bc930831ee9d45ea58b)
FetchContent_Declare(benchmark GIT_REPOSITORY https://github.com/google/benchmark.git GIT_TAG eddb0241389718a23a42db6af5f0164b6e0139af)
FetchContent_MakeAvailable(googletest catch2 benchmark)
if(NOT LAB3_BOOST_ROOT)
  FetchContent_Declare(boost_headers SOURCE_SUBDIR __headers_only__ URL https://archives.boost.io/release/1.88.0/source/boost_1_88_0.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(boost_headers)
  set(LAB3_BOOST_ROOT "${boost_headers_SOURCE_DIR}")
endif()
if(EXISTS "${LAB3_BOOST_ROOT}/boost")
  set(BOOST_INCLUDES "${LAB3_BOOST_ROOT}")
else()
  file(GLOB_RECURSE BOOST_CONFIGURE_HEADERS "${LAB3_BOOST_ROOT}/libs/*/include/boost/config.hpp")
  file(GLOB BOOST_MODULE_DIRS "${LAB3_BOOST_ROOT}/libs/*" "${LAB3_BOOST_ROOT}/libs/*/*")
  foreach(module ${BOOST_MODULE_DIRS})
    if(IS_DIRECTORY "${module}/include")
      list(APPEND BOOST_INCLUDES "${module}/include")
    endif()
  endforeach()
endif()
file(GLOB ADT_SOURCES CONFIGURE_DEPENDS str/*.cpp)
add_library(adt STATIC ${ADT_SOURCES})
target_include_directories(adt PUBLIC str)
target_compile_options(adt PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(LAB3_COVERAGE)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "Use Clang for LAB3_COVERAGE")
  endif()
  target_compile_options(adt PUBLIC -O0 -g -fprofile-instr-generate -fcoverage-mapping)
  target_link_options(adt PUBLIC -fprofile-instr-generate)
endif()
if(LAB3_SANITIZERS)
  target_compile_options(adt PUBLIC -g -fsanitize=${LAB3_SANITIZER_SET} -fno-omit-frame-pointer)
  target_link_options(adt PUBLIC -fsanitize=${LAB3_SANITIZER_SET})
endif()
add_executable(lab3 main.cpp)
target_link_libraries(lab3 PRIVATE adt)
file(GLOB GTEST_SOURCES CONFIGURE_DEPENDS tests/gtest/*.cpp)
add_executable(test_gtest ${GTEST_SOURCES})
target_link_libraries(test_gtest PRIVATE adt GTest::gtest_main)
add_executable(test_boost tests/boost/test_boost.cpp)
target_include_directories(test_boost SYSTEM PRIVATE ${BOOST_INCLUDES})
target_link_libraries(test_boost PRIVATE adt)
add_executable(test_catch2 tests/catch2/test_catch2.cpp)
target_link_libraries(test_catch2 PRIVATE adt Catch2::Catch2WithMain)
add_executable(benchmarks benchmarks/bench.cpp)
add_library(adt_bench STATIC ${ADT_SOURCES})
target_include_directories(adt_bench PUBLIC str)
target_compile_options(adt_bench PRIVATE -O2 -Wall -Wextra -Wpedantic -Werror)
target_compile_options(benchmarks PRIVATE -O2)
target_link_libraries(benchmarks PRIVATE adt_bench benchmark::benchmark)
enable_testing()
foreach(framework gtest boost catch2)
  add_test(NAME ${framework} COMMAND test_${framework})
  set_tests_properties(${framework} PROPERTIES ENVIRONMENT "LLVM_PROFILE_FILE=${CMAKE_BINARY_DIR}/${framework}-%p.profraw")
endforeach()
